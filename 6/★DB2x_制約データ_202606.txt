/* DB2x　表の制約（2026年6月）
   DBに以下の２つの表を作成し、
　 外部キー制約の動きを確認しよう
*/

/* T_Goods 作成　親表 */
create table T_Goods(
  F_Gid char(3),
  F_Gname varchar(10) not null,
  primary key(F_Gid)
);

/* 親表　データ入力　*/
insert into T_Goods values('100','Apple');
insert into T_Goods values('200','Book');
insert into T_Goods values('300','Pen');
commit;

/* T_Sales 作成子表　*/
create table T_Sales(
  F_SID  int(3),
  F_GID  char(3),
  F_DAY  date,
primary key(F_SID),
  foreign key(F_GID)
              references T_Goods(F_Gid)
);

/* 子表　データ入力　*/
insert into T_Sales values(2,'100','2025/05/26');
insert into T_Sales values(1,'200','2025/06/30');
commit;
