DROP TABLE T_Assignments;
DROP TABLE T_Course;
DROP TABLE T_Instructors;


/* 1番目の表作成 */

CREATE TABLE T_Course
(
F_CourseCode   CHAR(4)       NOT NULL,
F_CourseName   VARCHAR(100)  NOT NULL,
F_CourseFee    INT           NOT NULL,
F_EventPeriod  INT           NOT NULL,
Primary key(F_CourseCode)
);

/* テーブル作成2 */

CREATE TABLE T_Instructors
(
F_InstructorID  CHAR(4)      NOT NULL,
F_Name          VARCHAR(50)  NOT NULL,
F_PhoneNo       VARCHAR(20),
Primary key(F_InstructorID)
);

/* テーブル作成3 */

CREATE TABLE T_Assignments
(
F_CourseCode    CHAR(4)  NOT NULL,
F_InstructorID  CHAR(4)  NOT NULL,
Primary key(F_CourseCode, F_InstructorID),
Foreign key(F_CourseCode)   references T_Course(F_CourseCode),
Foreign key(F_InstructorID) references T_Instructors(F_InstructorID)
);
